library(tidyverse)
library(plotly)

full_data <- read_csv("../data/combined_data.csv", show_col_types = FALSE) |>
  filter(month >= as.Date("2024-12-01")) |>  # Exclude partial November 2024.
  arrange(month, State) |>
  mutate(
    month_label = format(month, "%Y-%m"),
    hover_label = paste0(
      State, "<br>Month: ", month_label,
      "<br>Hours for 15 gallons: ", round(hours_per_fillup, 2),
      "<br>Real gas price: $", round(MRA_Regular, 2), "/gallon"
    )
  )

1. Animated affordability map

plot_ly(
  data = full_data,
  type = "choropleth",
  locations = ~State,
  locationmode = "USA-states",
  z = ~hours_per_fillup,
  frame = ~month_label,
  text = ~hover_label,
  hoverinfo = "text",
  colorscale = "YlOrRd",
  zmin = min(full_data$hours_per_fillup),
  zmax = max(full_data$hours_per_fillup),
  colorbar = list(title = "Hours")
) |>
  layout(
    title = "Hours of Work Needed for 15 Gallons",
    geo = list(scope = "usa"),
    height = 600
  ) |>
  animation_opts(frame = 800, transition = 0, redraw = TRUE) |>
  animation_slider(currentvalue = list(prefix = "Month: "))

3. Average affordability over time

trend_data <- full_data |>
  group_by(month) |>
  summarise(
    mean_hours_per_fillup = mean(hours_per_fillup),
    .groups = "drop"
  )

trend_plot <- ggplot(
  trend_data,
  aes(x = month, y = mean_hours_per_fillup)
) +
  geom_line() +
  geom_point() +
  scale_x_date(date_breaks = "3 months", date_labels = "%b %Y") +
  labs(
    title = "Average Hours of Work Needed for 15 Gallons",
    subtitle = "Equal-weighted average across 50 states",
    x = NULL,
    y = "Hours of work"
  ) +
  theme_minimal() +
  theme(axis.text.x = element_text(angle = 35, hjust = 1))

ggplotly(trend_plot, tooltip = c("x", "y"))

October 2025 is absent because the CPI series used for the real variables is missing that month.
